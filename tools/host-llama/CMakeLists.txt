cmake_minimum_required(VERSION 3.22.1)
project(pararead_host_check LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
find_package(JNI REQUIRED)
add_subdirectory("${LLAMA_SOURCE}" "${CMAKE_BINARY_DIR}/llama")
# Compile the actual Android JNI implementation, without an independent inference wrapper.
add_library(pararead-llama SHARED ../../engine/llama/src/main/cpp/llama_jni.cpp)
target_include_directories(pararead-llama PRIVATE ${JNI_INCLUDE_DIRS} "${LLAMA_SOURCE}/include")
target_link_libraries(pararead-llama PRIVATE llama)
